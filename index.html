let questionBank=[];
let examQuestions=[];
let answers=[];
let current=0;
//=================================
function shuffle(array){
for(let i=array.length-1;i>0;i--){
const j=Math.floor(
Math.random()*(i+1)
);
[array[i],array[j]]=
[array[j],array[i]];
}
return array;
}
//=================================
function parseQuestion(text){
const regex=
/(.*)\(A\)(.*?)\(B\)(.*?)\(C\)(.*?)\(D\)(.*)/s;
const m=text.match(regex);
if(!m) return null;
return{
question:m[1].trim(),
A:m[2].trim(),
B:m[3].trim(),
C:m[4].trim(),
D:m[5].trim()
};
}
function downloadReview(){
let resultText = "";
examQuestions.forEach((q,i)=>{
resultText += `
================================
第 ${i+1} 題
題目：
${q.q}
A. ${q.A}
B. ${q.B}
C. ${q.C}
D. ${q.D}
您的答案：
${answers[i]
? answers[i] + ". " + q[answers[i]]
: "未作答"}
正確答案：
${q.answer}. ${q[q.answer]}
`;
});
const blob =
new Blob(
[resultText],
{
type:"text/plain;charset=utf-8"
}
);
const a =
document.createElement("a");
a.href =
URL.createObjectURL(blob);
a.download =
"模擬測驗結果.txt";
a.click();
}
//=================================
async function loadQuestions(){
const response = await fetch("questions.csv");
const csv = await response.text();
const rows = csv.split(/\r?\n/);
questionBank = [];
for(let i=1;i<rows.length;i++){
const row = rows[i].trim();
if(!row) continue;
const cols = row.split(",");
if(cols.length < 5) continue;
const answer =
cols[cols.length-1].trim();
const questionText =
cols.slice(2, cols.length-2)
.join(",");
const p = parseQuestion(questionText);
if(!p) continue;
questionBank.push({
q:p.question,
A:p.A,
B:p.B,
C:p.C,
D:p.D,
answer:answer
});
}
console.log(
"題庫載入完成：",
questionBank.length
);
examQuestions =
shuffle([...questionBank])
.slice(0,100);
answers =
new Array(100).fill(null);
renderQuestion();
}
//=================================
function renderQuestion(){
const q=
examQuestions[current];
document.getElementById("qno").innerText=
current+1;
document.getElementById("question").innerText=
(current+1)+". "+q.q;
let html="";
["A","B","C","D"]
.forEach(opt=>{
html+=`
<div class="option">
<label>
<input
type="radio"
name="answer"
value="${opt}"
${answers[current]===opt
?"checked":""
}
onchange="saveAnswer('${opt}')">
${opt}. ${q[opt]}
</label>
</div>
`;
});
document.getElementById("options")
.innerHTML=html;
updateCounter();
}
//=================================
function saveAnswer(ans){
answers[current]=ans;
updateCounter();
}
function updateCounter(){
document.getElementById("answered")
.innerText=
answers.filter(
x=>x!==null
).length;
}
//=================================
function nextQuestion(){
if(current<99){
current++;
renderQuestion();
}
}
function prevQuestion(){
if(current>0){
current--;
renderQuestion();
}
}
//=================================
let remain=
90*60;
function updateTimer(){
const h=
Math.floor(remain/3600);
const m=
Math.floor(
(remain%3600)/60
);
const s=
remain%60;
document.getElementById("timer")
.innerText=
`${String(h).padStart(2,"0")}:${String(m).padStart(2,"0")}:${String(s).padStart(2,"0")}`;
}
updateTimer();
const timer=
setInterval(()=>{
remain--;
updateTimer();
if(remain<=0){
clearInterval(timer);
alert("測驗時間到，自動交卷");
submitExam();
}
},1000);
//=================================
function submitExam(){
const confirmSubmit = confirm(
"確定要交卷嗎？\n\n交卷後將無法再修改答案。"
);
if(!confirmSubmit){
return;
}
clearInterval(timer);
let score=0;
let review="";
examQuestions.forEach((q,i)=>{
const userAns = answers[i];
const isCorrect =
userAns === q.answer;
if(isCorrect){
score++;
}
review += `
<div class="wrong-card">
<h3>
${isCorrect ? "✅" : "❌"}
第 ${i+1} 題
</h3>
<p>
<strong>題目：</strong>
${q.q}
</p>
<div class="wrong-answer">
<strong>您的答案：</strong>
${
userAns
? `${userAns}. ${q[userAns]}`
: "未作答"
}
</div>
<div class="correct-answer">
<strong>正確答案：</strong>
${q.answer}. ${q[q.answer]}
</div>
</div>
`;
review += `
<div class="wrong-card">
<h3>❌ 第 ${i+1} 題</h3>
<p><strong>題目：</strong>${q.q}</p>
<div class="wrong-answer">
<strong>您的答案：</strong>
${userAns? `${userAns}. ${q[userAns]}`: "未作答"}
</div>
<div class="correct-answer">
<strong>正確答案：</strong>
${q.answer}. ${q[q.answer]}
</div>
</div>
`;
}
});
document.getElementById("result")
.innerHTML=`
<h2>測驗完成</h2>
<h1>${score} 分</h1>
<h2 style="
color:${score>=60?'green':'red'};
">
${score>=60?'✅及格':'❌不及格'}
</h2>
<p>答對：${score} 題</p>
<p>答錯：${100-score} 題</p>
<hr>
<h3>完整試卷與答案</h3>
${review}
<button
class="btn-primary"
onclick="downloadReview()">
📥下載完整考卷
</button>
<button
class="btn-success"
onclick="location.reload()">
重新測驗
</button>
`;
document.getElementById("examArea")
.style.display="none";
document.getElementById("result")
.style.display="block";
window.scrollTo(0,0);
}
